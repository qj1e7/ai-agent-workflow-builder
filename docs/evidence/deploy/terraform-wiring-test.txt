# P14 test of infra/modules/cosmos with a mocked provider. Nothing contacts Azure.
mock_provider "azurerm" {
  mock_resource "azurerm_cosmosdb_account" {
    defaults = { id = "/subscriptions/00000000-0000-0000-0000-000000000000/resourceGroups/rg-wab-dev/providers/Microsoft.DocumentDB/databaseAccounts/cosmos-wab-dev-abcde" }
  }
}

variables {
  name                = "cosmos-wab-dev-abcde"
  location            = "southeastasia"
  resource_group_name = "rg-wab-dev"
  serverless          = false
  data_contributors   = { platform-api = "11111111-1111-1111-1111-111111111111" }
  tags                = { project = "ai-workflow-builder" }
}

run "free_tier" {
  command = apply

  assert {
    condition     = azurerm_cosmosdb_account.this.free_tier_enabled && length(azurerm_cosmosdb_account.this.capabilities) == 0
    error_message = "the default account is on the free tier, not serverless"
  }
  assert {
    condition     = azurerm_cosmosdb_account.this.capacity[0].total_throughput_limit == 1000
    error_message = "the account is capped at the free 1000 RU/s"
  }
  assert {
    condition     = azurerm_cosmosdb_account.this.consistency_policy[0].consistency_level == "Strong"
    error_message = "Strong consistency"
  }
  assert {
    condition     = length(azurerm_cosmosdb_account.this.geo_location) == 1 && one(azurerm_cosmosdb_account.this.geo_location).location == "southeastasia"
    error_message = "one region"
  }
  assert {
    condition     = azurerm_cosmosdb_account.this.access_key_metadata_writes_enabled == false
    error_message = "the key cannot create or change databases and containers"
  }
  assert {
    condition     = azurerm_cosmosdb_sql_database.activity.name == "activity" && azurerm_cosmosdb_sql_container.run_events.name == "run_events"
    error_message = "the names platform-api's adapter uses"
  }
  assert {
    condition     = azurerm_cosmosdb_sql_container.run_events.partition_key_paths == tolist(["/runId"]) && azurerm_cosmosdb_sql_container.run_events.partition_key_version == 2
    error_message = "partitioned by run id, as the SDK's default version 2"
  }
  assert {
    condition     = azurerm_cosmosdb_sql_container.run_events.throughput == 1000
    error_message = "the container uses the free 1000 RU/s"
  }
  assert {
    condition     = endswith(azurerm_cosmosdb_sql_role_assignment.data_contributor["platform-api"].role_definition_id, "/sqlRoleDefinitions/00000000-0000-0000-0000-000000000002")
    error_message = "the built-in Data Contributor role"
  }
  assert {
    condition     = endswith(azurerm_cosmosdb_sql_role_assignment.data_contributor["platform-api"].scope, "/dbs/activity")
    error_message = "scoped to the activity database"
  }
  assert {
    condition     = azurerm_cosmosdb_sql_role_assignment.data_contributor["platform-api"].principal_id == "11111111-1111-1111-1111-111111111111"
    error_message = "given to platform-api's identity"
  }
}

run "serverless" {
  command = apply

  variables {
    serverless = true
  }

  assert {
    condition     = !azurerm_cosmosdb_account.this.free_tier_enabled
    error_message = "serverless has no free tier"
  }
  assert {
    condition     = [for c in azurerm_cosmosdb_account.this.capabilities : c.name] == ["EnableServerless"]
    error_message = "the serverless capability"
  }
  assert {
    condition     = length(azurerm_cosmosdb_account.this.capacity) == 0
    error_message = "no throughput cap on a serverless account"
  }
}
